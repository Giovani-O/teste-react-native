### Task 8: Create School List Screen

**Files:**
- Create: `src/screens/schools/SchoolListScreen.tsx`

- [ ] **Step 1: Create SchoolListScreen component**

```typescript
import { Box } from '@gluestack-ui/ui'
import { Text } from '@gluestack-ui/ui'
import { VStack } from '@gluestack-ui/ui'
import { HStack } from '@gluestack-ui/ui'
import { Pressable } from '@gluestack-ui/ui'
import { useRouter } from 'expo-router'
import { useState } from 'react'
import { mockSchools } from '../../data/mocks/mockData'
import { School } from '../../domain/entities/School'
import { FAB } from '../../components/FAB'
import { BottomSheet } from '../../components/BottomSheet'

export function SchoolListScreen() {
  const router = useRouter()
  const [schools] = useState<School[]>(mockSchools)
  const [filterOpen, setFilterOpen] = useState(false)
  const [createOpen, setCreateOpen] = useState(false)

  const handleSchoolPress = (id: string) => {
    router.push(`/schools/${id}`)
  }

  return (
    <Box flex={1} bg="$white">
      {schools.length === 0 ? (
        <VStack flex={1} justifyContent="center" alignItems="center" p={8}>
          <Text fontSize="$xl" color="$textSecondary">Nenhuma escola encontrada</Text>
          <Text fontSize="$sm" color="$textMuted" mt={2}>Toque no botão + para adicionar uma escola</Text>
        </VStack>
      ) : (
        <VStack borderTopWidth={1} borderTopColor="$borderLight100">
          {schools.map((school) => (
            <Pressable
              key={school.id}
              onPress={() => handleSchoolPress(school.id)}
              borderBottomWidth={1}
              borderBottomColor="$borderLight100"
              py={4}
              px={4}
              bg="$white"
              _pressed={{ bg: '$backgroundLight100' }}
            >
              <VStack gap={1}>
                <Text fontSize="$md" fontWeight="$semibold" color="$textPrimary">
                  {school.name}
                </Text>
                <Text fontSize="$sm" color="$textSecondary">
                  {school.address}
                </Text>
                <Text fontSize="$xs" color="$textMuted" mt={1}>
                  {school.classCount} {school.classCount === 1 ? 'turma' : 'turmas'}
                </Text>
              </VStack>
            </Pressable>
          ))}
        </VStack>
      )}

      <FAB onPress={() => setCreateOpen(true)} />

      <BottomSheet isOpen={filterOpen} onClose={() => setFilterOpen(false)}>
        <VStack p={4} gap={4}>
          <Text fontSize="$lg" fontWeight="$semibold">Filtros</Text>
          <Text size="sm" color="$textSecondary">Em construção...</Text>
        </VStack>
      </BottomSheet>

      <BottomSheet isOpen={createOpen} onClose={() => setCreateOpen(false)}>
        <VStack p={4} gap={4}>
          <Text fontSize="$lg" fontWeight="$semibold">Nova Escola</Text>
          <Text size="sm" color="$textSecondary">Em construção...</Text>
        </VStack>
      </BottomSheet>
    </Box>
  )
}
```

### Task 9: Create School Detail Screen

**Files:**
- Create: `src/screens/schools/SchoolDetailScreen.tsx`

- [ ] **Step 1: Create SchoolDetailScreen component**

```typescript
import { Box } from '@gluestack-ui/ui'
import { Text } from '@gluestack/ui'
import { VStack } from '@gluestack/ui'
import { HStack } from '@gluestack/ui'
import { Pressable } from '@gluestack/ui'
import { Badge, BadgeText } from '@gluestack-ui/ui'
import { useLocalSearchParams } from 'expo-router'
import { useState, useEffect } from 'react'
import { mockSchools, mockTurmas } from '../../data/mocks/mockData'
import { School } from '../../domain/entities/School'
import { Turma, Shift } from '../../domain/entities/Turma'
import { FAB } from '../../components/FAB'
import { BottomSheet } from '../../components/BottomSheet'

const shiftLabels: Record<Shift, string> = {
  morning: 'Manhã',
  afternoon: 'Tarde',
  evening: 'Noite',
}

export function SchoolDetailScreen() {
  const { id } = useLocalSearchParams<{ id: string }>()
  const [school, setSchool] = useState<School | null>(null)
  const [turmas, setTurmas] = useState<Turma[]>([])
  const [filterOpen, setFilterOpen] = useState(false)
  const [createOpen, setCreateOpen] = useState(false)

  useEffect(() => {
    const foundSchool = mockSchools.find((s) => s.id === id)
    setSchool(foundSchool || null)
    setTurmas(mockTurmas[id] || [])
  }, [id])

  if (!school) {
    return (
      <Box flex={1} justifyContent="center" alignItems="center">
        <Text>Escola não encontrada</Text>
      </Box>
    )
  }

  return (
    <Box flex={1} bg="$white">
      <VStack p={4} borderBottomWidth={1} borderBottomColor="$borderLight100" gap={2}>
        <Text fontSize="$xl" fontWeight="$bold" color="$textPrimary">
          {school.name}
        </Text>
        <Text fontSize="$sm" color="$textSecondary">
          {school.address}
        </Text>
        <Text fontSize="$xs" color="$textMuted" mt={2}>
          {school.classCount} {school.classCount === 1 ? 'turma cadastrada' : 'turmas cadastradas'}
        </Text>
      </VStack>

      <Text fontSize="$md" fontWeight="$semibold" p={4} pb={2} color="$textPrimary">
        Turmas
      </Text>

      {turmas.length === 0 ? (
        <VStack flex={1} justifyContent="center" alignItems="center" p={8}>
          <Text fontSize="$md" color="$textSecondary">Nenhuma turma encontrada</Text>
          <Text fontSize="$sm" color="$textMuted" mt={2}>Toque no botão + para adicionar uma turma</Text>
        </VStack>
      ) : (
        <VStack borderTopWidth={1} borderTopColor="$borderLight100">
          {turmas.map((turma) => (
            <Pressable
              key={turma.id}
              borderBottomWidth={1}
              borderBottomColor="$borderLight100"
              py={4}
              px={4}
              bg="$white"
              _pressed={{ bg: '$backgroundLight100' }}
            >
              <HStack justifyContent="space-between" alignItems="center">
                <VStack gap={1}>
                  <Text fontSize="$md" fontWeight="$medium" color="$textPrimary">
                    {turma.name}
                  </Text>
                  <Text fontSize="$sm" color="$textSecondary">
                    {turma.academicYear}
                  </Text>
                </VStack>
                <Badge
                  size="sm"
                  variant="solid"
                  bg={turma.shift === 'morning' ? '#0066CC' : turma.shift === 'afternoon' ? '#7C3AED' : '#059669'}
                >
                  <BadgeText color="white">{shiftLabels[turma.shift]}</BadgeText>
                </Badge>
              </HStack>
            </Pressable>
          ))}
        </VStack>
      )}

      <FAB onPress={() => setCreateOpen(true)} />

      <BottomSheet isOpen={filterOpen} onClose={() => setFilterOpen(false)}>
        <VStack p={4} gap={4}>
          <Text fontSize="$lg" fontWeight="$semibold">Filtrar Turmas</Text>
          <Text size="sm" color="$textSecondary">Em construção...</Text>
        </VStack>
      </BottomSheet>

      <BottomSheet isOpen={createOpen} onClose={() => setCreateOpen(false)}>
        <VStack p={4} gap={4}>
          <Text fontSize="$lg" fontWeight="$semibold">Nova Turma</Text>
          <Text size="sm" color="$textSecondary">Em construção...</Text>
        </VStack>
      </BottomSheet>
    </Box>
  )
}
```
